# 正 N 边形(可带圆角)路径数据生成器:核心/星芒都用它,方便调形状
# 注意:PowerShell 里 @(a, b) 中逗号的优先级比 + * 高,元素里带运算必须拆成变量,
#       否则 @($CX + Cos($a)*$R, $CY + ...) 会被解析成 $CX + (Cos($a)*($R,$CY))
function New-RoundPolyData {
    param(
        [double]$R, [int]$N, [double]$Round = 0.4, [double]$Phase = 0,
        [double]$CX = 120, [double]$CY = 122
    )
    $v = @()
    for ($i = 0; $i -lt $N; $i++) {
        $a = 2 * [Math]::PI * $i / $N + $Phase
        $px = $CX + [Math]::Cos($a) * $R
        $py = $CY + [Math]::Sin($a) * $R
        $v += , @($px, $py)
    }
    $k = $Round
    $en = @(); $ex = @()
    for ($i = 0; $i -lt $N; $i++) {
        $vi = $v[$i]
        $pv = $v[($i - 1 + $N) % $N]
        $nv = $v[($i + 1) % $N]
        $en1 = $vi[0] + ($pv[0] - $vi[0]) * $k
        $en2 = $vi[1] + ($pv[1] - $vi[1]) * $k
        $en += , @($en1, $en2)
        $ex1 = $vi[0] + ($nv[0] - $vi[0]) * $k
        $ex2 = $vi[1] + ($nv[1] - $vi[1]) * $k
        $ex += , @($ex1, $ex2)
    }
    $sb = New-Object System.Text.StringBuilder
    [void]$sb.Append('M ' + (F $ex[0][0]) + ',' + (F $ex[0][1]))
    for ($i = 1; $i -lt $N; $i++) {
        [void]$sb.Append(' L ' + (F $en[$i][0]) + ',' + (F $en[$i][1]))
        [void]$sb.Append(' Q ' + (F $v[$i][0]) + ',' + (F $v[$i][1]) + ' ' + (F $ex[$i][0]) + ',' + (F $ex[$i][1]))
    }
    [void]$sb.Append(' L ' + (F $en[0][0]) + ',' + (F $en[0][1]))
    [void]$sb.Append(' Q ' + (F $v[0][0]) + ',' + (F $v[0][1]) + ' ' + (F $ex[0][0]) + ',' + (F $ex[0][1]))
    [void]$sb.Append(' Z')
    return $sb.ToString()
}

# 核心:六边形(尖朝上),圆角较大,像一个软软的坐垫
# 豆包 = 一颗豆: 圆角矩形拉长一点(152x120)、圆角很大、整体斜 8 度
function New-BeanData {
    param([double]$CX = 120, [double]$CY = 122, [double]$W = 152, [double]$H = 120, [double]$R = 56, [double]$Rot = -8)
    $hw = $W / 2; $hh = $H / 2
    $pts = @(
        @((- $hw + $R), (- $hh)), @(($hw - $R), (- $hh)), @($hw, (- $hh + $R)), @($hw, ($hh - $R)),
        @(($hw - $R), $hh), @((- $hw + $R), $hh), @((- $hw), ($hh - $R)), @((- $hw), (- $hh + $R))
    )
    $rad = $Rot * [Math]::PI / 180.0
    $co = [Math]::Cos($rad); $si = [Math]::Sin($rad)
    $abs = @()
    foreach ($p in $pts) {
        $x = $p[0] * $co - $p[1] * $si
        $y = $p[0] * $si + $p[1] * $co
        $abs += , @(($CX + $x), ($CY + $y))
    }
    $sb = New-Object System.Text.StringBuilder
    [void]$sb.Append(('M {0:0.##},{1:0.##} L {2:0.##},{3:0.##} ' -f $abs[0][0], $abs[0][1], $abs[1][0], $abs[1][1]))
    foreach ($k in 2, 4, 6) {
        [void]$sb.Append(('A {0:0.##},{0:0.##} 0 0 1 {1:0.##},{2:0.##} ' -f $R, $abs[$k][0], $abs[$k][1]))
        $nx = $k + 1
        [void]$sb.Append(('L {0:0.##},{1:0.##} ' -f $abs[$nx][0], $abs[$nx][1]))
    }
    [void]$sb.Append(('A {0:0.##},{0:0.##} 0 0 1 {1:0.##},{2:0.##} Z' -f $R, $abs[0][0], $abs[0][1]))
    return $sb.ToString()
}
